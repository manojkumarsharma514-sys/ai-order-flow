    def close_position_by_id(self, position_id):
        """Manual 'Close Position' button on the dedicated Positions tab."""

        result = self.paper_engine.close_position(position_id, reason="manual")
        if result is None:
            print(f"⚠️ close_position_by_id: no open position with id {position_id}")
        self.refresh_positions()

    def _on_sl_tp_updated(self, position_id, new_stop_loss, new_take_profit):
        """'✏ Edit' button on either Positions panel (compact dashboard
        tab or dedicated Positions tab) — apply the trader's edited
        Stop Loss / Take Profit to the live position. On rejection
        (e.g. the value is on the wrong side of the current mark price,
        or the position closed between opening the dialog and clicking
        Save), surface the reason in the Trade Setup panel's warning
        banner — same "never fail silently" pattern already used for
        auto-trade rejections and manual order rejections — rather than
        just logging it to the console where the trader would never see
        it."""

        ok, reason = self.paper_engine.modify_position_sl_tp(
            position_id, stop_loss=new_stop_loss, take_profit=new_take_profit
        )

        if not ok:
            print(f"⚠️ SL/TP edit rejected for position {position_id}: {reason}")
            try:
                self.dashboard.trade_setup.show_rejection(f"SL/TP edit failed: {reason}")
            except Exception as e:
                print(f"SL/TP rejection banner error: {e}")
            return

        self.refresh_positions()
